[source,bash]
----
git -C "$CORE" merge-base --is-ancestor "$ANCHOR" refs/heads/main || exit 1

  "$CORE" "$ANCHOR_SOURCE"
CLONED_TIP=$(git -C "$ANCHOR_SOURCE" rev-parse refs/heads/main)

disposable preparation input, never a provenance authority: Core plus the
extraction anchor remain authoritative, and filter-repo never runs there.
